# One command per line. Lines starting with # are skipped.
# Bootstrap.sh's seed-history module appends these to ~/.bash_history (deduped),
# then runs `atuin import bash`. After that, Ctrl-R fuzzy-finds them in atuin's TUI.

# "I need to check docker memory", not "scenario 06's command."

# ============================================================================
# ===== SYMPTOM QUICK TRIAGE — first 3 commands per symptom =====
# ============================================================================
# Use this as your "hands first, MCP second" reference.
# Run these BEFORE hitting MCP. Narrate as you go.
#
# --- container won't start / keeps restarting ---
#   → check exit code + OOM + restart count, then read last logs
docker inspect <c> --format '{{.State.Status}} exit={{.State.ExitCode}} oom={{.State.OOMKilled}} restarts={{.RestartCount}}'
docker logs --tail 30 <c> 2>&1
docker events --since 10m --filter event=die --filter event=oom --format '{{.Time}} {{.Action}} {{.Actor.Attributes.name}}'
#
# --- DNS failure (ENOTFOUND / EAI_AGAIN / resolution slow) ---
#   → check resolv.conf first (ndots, search domains), then test resolution, then /etc/hosts
cat /etc/resolv.conf
dig +short <host> || nslookup <host>
cat /etc/hosts | grep -v '^#'
#
# --- TLS / cert error (UNABLE_TO_VERIFY / CERT_HAS_EXPIRED / unknown authority) ---
#   → check what cert the server sends, check dates, check trust store env vars
openssl s_client -connect <host>:443 < /dev/null 2>&1 | grep -E 'subject|issuer|verify|depth'
openssl s_client -connect <host>:443 < /dev/null 2>&1 | openssl x509 -noout -dates
echo $NODE_EXTRA_CA_CERTS $REQUESTS_CA_BUNDLE $SSL_CERT_FILE
#
# --- high CPU / process runaway ---
#   → find the hot process, check its threads, check if it's throttled
ps -eo pid,pcpu,rss,comm --sort=-pcpu | head -10
top -H -p <pid> -bn1 | head -20
cat /sys/fs/cgroup/cpu.stat | grep -E 'nr_throttled|throttled_usec'
#
# --- disk full / ENOSPC ---
#   → check filesystem usage, then find big files, check docker disk
df -h
du -sh /* 2>/dev/null | sort -rh | head -10
docker system df
#
# --- OOMKilled / exit 137 ---
#   → confirm OOM in dmesg, check which process got killed, check cgroup memory limit
sudo dmesg -T | grep -i 'killed process\|out of memory' | tail -5
docker inspect <c> --format '{{.State.OOMKilled}} exit={{.State.ExitCode}} mem_limit={{.HostConfig.Memory}}'
cat /proc/meminfo | grep -E 'MemTotal|MemAvailable|SwapTotal|SwapFree'
#
# --- port conflict / EADDRINUSE ---
#   → find what's on the port, check if it's the expected process
ss -tlnp 'sport = :<port>'
sudo lsof -i :<port>
docker port <c>
#
# --- permission denied / EACCES ---
#   → check file perms, trace the full path, check if SELinux/AppArmor is blocking
ls -la /path/to/file
namei -l /path/to/file
sudo dmesg | grep -i 'denied\|apparmor\|audit' | tail -10
#
# --- zombie processes (Z state) ---
#   → find zombies, check what PID 1 is (can it reap?), check parent
ps -eo state,pid,ppid,cmd | awk '$1=="Z"'
ps -p 1 -o pid,comm
ps -eo pid,ppid,stat,cmd | awk '$3 ~ /Z/ {print "zombie PID="$1, "parent="$2}'
#
# --- slow / hanging process (D state) ---
#   → check for D-state processes, check I/O wait, check what syscall it's stuck on
ps -eo state,pid,wchan,cmd | awk '$1=="D"'
cat /proc/pressure/io
sudo cat /proc/<pid>/stack
#
# --- memory leak / growing RSS ---
#   → snapshot RSS, wait, snapshot again, check swap
ps -eo pid,rss,vsz,comm --sort=-rss | head -10
free -h
cat /proc/<pid>/status | grep -E 'VmRSS|VmSwap|VmSize'
#
# --- fd exhaustion / EMFILE / too many open files ---
#   → check fd count vs limit for the process
ls /proc/<pid>/fd | wc -l
cat /proc/<pid>/limits | grep 'Max open files'
lsof -p <pid> | wc -l
#
# --- systemd unit failed ---
#   → check status, read the logs, check dependencies
systemctl status <unit>
journalctl -u <unit> --since '10 min ago' --no-pager
systemctl list-dependencies <unit>
#
# --- container networking broken (can't reach other containers) ---
#   → check what network the container is on, check iptables FORWARD, test connectivity
docker inspect <c> --format '{{json .NetworkSettings.Networks}}' | jq
sudo iptables -L FORWARD -n -v --line-numbers
docker exec <c> nc -zv <target-host> <port>
#
# --- swap thrashing / system unresponsive ---
#   → check swap usage, find swap consumers, check memory pressure
swapon --show
vmstat 1 5
cat /proc/pressure/memory
#
# --- clock skew (token expired / cert valid but rejected) ---
#   → compare local time to remote, check NTP
date -u
curl -sI https://www.google.com | grep -i ^date
timedatectl
#
# --- config file mismatch (app not reading expected config) ---
#   → check what config the process actually loaded, check env vars
cat /proc/<pid>/cmdline | tr '\0' ' '
cat /proc/<pid>/environ | tr '\0' '\n' | grep -iE 'config|conf|setting'
docker inspect <c> --format '{{json .Config.Env}}' | jq
