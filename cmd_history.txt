# ===== Linux debugging =====
sudo dmesg -T | tail -50
sudo dmesg -T | grep -i 'killed process\|out of memory'
ps -o pid,stat,wchan,cmd | awk '$2 ~ /^D/'
sudo cat /proc/<pid>/stack
sudo cat /proc/<pid>/wchan
cat /proc/<pid>/status | grep -E 'State|VmRSS|Cap'
free -h
vmstat 1 10
iostat -xz 1 5
mpstat -P ALL 1 3
sar -n DEV 1 5
ss -tnp
nstat -az | grep -E 'TcpExt|IpExt'
sudo perf stat -p $(pgrep -f myapp) -- sleep 10
sudo perf record -F 99 -g --call-graph dwarf -p $(pgrep -f myapp) -- sleep 30
sudo bpftrace -e 'tracepoint:syscalls:sys_enter_openat { @[comm,args->filename] = count(); }'
sudo offcputime-bpfcc -p $(pgrep -f myapp) 30
